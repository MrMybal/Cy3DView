function(cy_dependency name url hash)
    if(EXISTS "${CMAKE_SOURCE_DIR}/third_party/${name}")
        FetchContent_Declare(${name} SOURCE_DIR "${CMAKE_SOURCE_DIR}/third_party/${name}")
    else()
        FetchContent_Declare(${name} URL "${url}" URL_HASH "SHA256=${hash}")
    endif()
endfunction()
